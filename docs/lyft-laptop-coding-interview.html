P01. Transactional Key-Value Store

写出 set/get/delete/begin/rollback/commit；解释 snapshot 的 O(n) 复制成本。

关键语法：dict · snapshots · transactions

展开 Python 参考实现 + assert 测试
class Database:
    def __init__(self):
        self.data = {}
        self.history = []

    def set(self, key, value):
        self.data[key] = value

    def get(self, key):
        return self.data.get(key)

    def delete(self, key):
        self.data.pop(key, None)

    def begin(self):
        self.history.append(self.data.copy())

    def rollback(self):
        if not self.history:
            raise RuntimeError("No transaction")
        self.data = self.history.pop()

    def commit(self):
        if not self.history:
            raise RuntimeError("No transaction")
        self.history.pop()

db = Database()
db.set("a", 1)
db.begin()
db.set("a", 2)
db.rollback()
assert db.get("a") == 1
db.begin()
db.set("b", 3)
db.commit()
assert db.get("b") == 3

P02. Priority Job Scheduler

最大优先级先执行；相同优先级保持 FIFO。

关键语法：heapq · itertools.count

展开 Python 参考实现 + assert 测试
import heapq
from itertools import count

class Scheduler:
    def __init__(self):
        self.heap = []
        self.seq = count()

    def add(self, priority, job):
        heapq.heappush(self.heap, (-priority, next(self.seq), job))

    def pop(self):
        if not self.heap:
            return None
        return heapq.heappop(self.heap)[2]

s = Scheduler()
s.add(1, "low")
s.add(5, "first")
s.add(5, "second")
assert [s.pop(), s.pop(), s.pop(), s.pop()] == [
    "first", "second", "low", None
]

P03. Pagination API Wrapper

逐页获取、按 id 去重、检测重复 cursor 防止死循环。

关键语法：loop · cursor · set · validation

展开 Python 参考实现 + assert 测试
def fetch_all(fetch_page):
    results, ids, cursors = [], set(), set()
    cursor = None
    while True:
        page = fetch_page(cursor)
        if not isinstance(page, dict):
            raise ValueError("Invalid page")
        for item in page.get("items", []):
            if item["id"] not in ids:
                ids.add(item["id"])
                results.append(item)
        next_cursor = page.get("next")
        if next_cursor is None:
            return results
        if next_cursor in cursors:
            raise ValueError("Cursor loop")
        cursors.add(next_cursor)
        cursor = next_cursor

pages = {
    None: {"items": [{"id": 1}], "next": "p2"},
    "p2": {"items": [{"id": 1}, {"id": 2}], "next": None}
}
assert len(fetch_all(lambda c: pages[c])) == 2

P04. Time-Based Key-Value Store — 乱序 timestamp

升级：不要求 timestamp 递增；同一 key、同一 timestamp 再次写入时覆盖旧值。get(key, t) 返回 timestamp ≤ t 的最近值，若不存在则返回空字符串。

推荐实现：dict + sorted timestamp list + bisect.insort；新时间戳 set O(n)，重复时间戳覆盖平均 O(1)，get O(log n)。列表插入需要移动元素，因此 insort 不是 O(log n) 写入。

乱序 timestamp：实时有序索引
from bisect import bisect_right, insort

class TimeMap:
    def __init__(self):
        self.times = {}   # key -> sorted timestamps
        self.values = {}  # key -> {timestamp: value}

    def set(self, key, value, timestamp):
        if key not in self.values:
            self.values[key] = {}
            self.times[key] = []
        if timestamp not in self.values[key]:
            insort(self.times[key], timestamp)
        self.values[key][timestamp] = value

    def get(self, key, timestamp):
        if key not in self.times:
            return ""
        times = self.times[key]
        i = bisect_right(times, timestamp) - 1
        if i < 0:
            return ""
        return self.values[key][times[i]]

tm = TimeMap()
tm.set("foo", "v10", 10)
tm.set("foo", "v3", 3)
tm.set("foo", "v7", 7)
tm.set("foo", "updated", 7)
assert tm.get("foo", 8) == "updated"
assert tm.get("foo", 5) == "v3"
assert tm.get("foo", 2) == ""
assert tm.get("foo", 10) == "v10"
assert tm.get("missing", 100) == ""

面试解释：如果第三方 SortedContainers 可用，可以使用 SortedDict / SortedList；但不能假设面试环境安装了额外依赖。Python 标准库的 bisect 提供二分查找，但 list 插入仍为 O(n)。

P15. Command Parser + JSON File I/O

将命令列表转为数据库操作，处理未知命令、缺少字段和文件输入输出。

关键语法：json · with open · try/except

展开 Python 参考实现 + assert 测试
import json

def execute(commands):
    db, output = {}, []
    for cmd in commands:
        try:
            op = cmd["op"].upper()
            if op == "SET":
                db[cmd["key"]] = cmd["value"]
            elif op == "GET":
                output.append(db.get(cmd["key"]))
            elif op == "DELETE":
                db.pop(cmd["key"], None)
            else:
                raise ValueError(f"Unknown operation: {op}")
        except (KeyError, TypeError, AttributeError) as exc:
            raise ValueError("Malformed command") from exc
    return output

def process_file(input_file, output_file):
    with open(input_file, encoding="utf-8") as f:
        commands = json.load(f)
    result = execute(commands)
    with open(output_file, "w", encoding="utf-8") as f:
        json.dump(result, f, ensure_ascii=False, indent=2)

assert execute([
    {"op": "SET", "key": "x", "value": 42},
    {"op": "GET", "key": "x"}
]) == [42]

P16. Merge Intervals

按起点排序并合并重叠区间，O(n log n)。

关键语法：sorted · greedy · max

展开 Python 参考实现 + assert 测试
def merge_intervals(intervals):
    if not intervals:
        return []
    intervals = sorted(intervals, key=lambda x: x[0])
    merged = [list(intervals[0])]
    for start, end in intervals[1:]:
        if start <= merged[-1][1]:
            merged[-1][1] = max(merged[-1][1], end)
        else:
            merged.append([start, end])
    return merged

assert merge_intervals([[1, 3], [2, 6], [8, 10]]) == [
    [1, 6], [8, 10]
]

P07. Find Median from Data Stream — Two Heaps

一个 Max Heap 保存较小一半，一个 Min Heap 保存较大一半。插入 O(log n)，查询中位数 O(1)。

Python heapq 只有 Min Heap；存负数实现 Max Heap。关键是不变量：low 的数量等于 high 或多一个。

展开 Python 实现 + assert 测试
import heapq

class MedianFinder:
    def __init__(self):
        self.low = []   # max heap simulated with negatives
        self.high = []  # min heap

    def add_num(self, num):
        heapq.heappush(self.low, -num)
        heapq.heappush(self.high, -heapq.heappop(self.low))
        if len(self.high) > len(self.low):
            heapq.heappush(self.low, -heapq.heappop(self.high))

    def find_median(self):
        if not self.low:
            raise ValueError("No numbers")
        if len(self.low) > len(self.high):
            return -self.low[0]
        return (-self.low[0] + self.high[0]) / 2

m = MedianFinder()
for x, expected in [(1, 1), (5, 3), (2, 2), (10, 3.5)]:
    m.add_num(x)
    assert m.find_median() == expected

P08. Lyft Minimum Workers with Assignment History — Two Min Heaps

候选人面经相关：输入每个 Job 的开始时间与持续时间；最少 Worker 覆盖所有 Job，返回按原始输入顺序排列的 Worker ID。空闲 Worker 多于一个时，优先使用最小 ID。任务结束时间等于新任务开始时间时允许复用。

两个 Min Heap：busy 保存 (end_time, worker_id)，available 保存 worker_id。先释放所有 end_time ≤ start 的 Worker，再复用最小 ID，否则创建新 Worker。关键：排序时保留原始 index；HHMM 必须先转分钟，不能直接做整数加法。

排序与堆操作总体 O(n log n) 时间；O(n) 空间。下方包括分钟版与 HHMM 输入适配版。要求 duration > 0，跨午夜任务以绝对分钟表示或单独约定。

展开 Lyft Worker Assignment Python 实现 + 测试
import heapq

def assign_workers(jobs):
    # jobs[i] = (start_minute, duration_minute)
    busy = []       # min heap: (end_time, worker_id)
    available = []  # min heap: worker_id
    assignment = [0] * len(jobs)
    next_worker_id = 0

    order = sorted(range(len(jobs)), key=lambda i: (jobs[i][0], i))
    for i in order:
        start, duration = jobs[i]
        if duration <= 0:
            raise ValueError("Duration must be positive")

        while busy and busy[0][0] <= start:
            _, worker_id = heapq.heappop(busy)
            heapq.heappush(available, worker_id)

        if available:
            worker_id = heapq.heappop(available)
        else:
            worker_id = next_worker_id
            next_worker_id += 1

        assignment[i] = worker_id
        heapq.heappush(busy, (start + duration, worker_id))

    return next_worker_id, assignment

def hhmm_to_minutes(hhmm):
    hours, minutes = divmod(hhmm, 100)
    if not (0 <= hours < 24 and 0 <= minutes < 60):
        raise ValueError("Invalid HHMM")
    return hours * 60 + minutes

def assign_workers_hhmm(jobs):
    return assign_workers([
        (hhmm_to_minutes(start_hhmm), duration)
        for start_hhmm, duration in jobs
    ])

assert assign_workers([
    (60, 30), (70, 10), (90, 15), (60, 5)
]) == (2, [0, 1, 0, 1])
assert assign_workers([]) == (0, [])
assert assign_workers([(0, 10), (10, 5)]) == (1, [0, 0])
assert assign_workers_hhmm([(1030, 30), (1100, 15)]) == (1, [0, 0])
